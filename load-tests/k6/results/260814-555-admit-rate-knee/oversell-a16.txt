# 실행: docker exec -i ticketrush-mysql mysql -uroot -p"$PW" ticket_rush -t -e "<아래 SQL>"
# 대상: performance_id = 35 (LTQ 코호트, 시딩 C=4,800)

-- (a1) 같은 좌석을 가리키는 예매 2건 이상 = oversell
SELECT seat_id, COUNT(*) c FROM booking WHERE performance_id=35 GROUP BY seat_id HAVING c > 1;
→ 0행

-- (a1-2) 1회 여정 가드 확인
SELECT user_id, COUNT(*) c FROM booking WHERE performance_id=35 GROUP BY user_id HAVING c > 1;
→ 0행

-- (a1-3) 총괄 — 셋이 같아야 한다.
SELECT COUNT(*) bookings, COUNT(DISTINCT seat_id) distinct_seats,
       COUNT(DISTINCT user_id) distinct_users
  FROM booking WHERE performance_id=35;
→ bookings=3842  distinct_seats=3842  distinct_users=3842   (일치)
