# SOLD 23,000 석은 hold_expired_at 이 NULL 로 정리됐고, 남은 HOLD 7,000 석의 만료 시각이
# 13:15:37Z 다. 이 시각 이후 인스턴스를 다시 켜면 기동 직후 첫 tick 에 한꺼번에 해제된다.
$ SELECT seat_status, COUNT(*) c, MIN(hold_expired_at), MAX(hold_expired_at)
    FROM seat WHERE performance_id=19 GROUP BY seat_status;

seat_status	c	min_exp	max_exp
SOLD	23000	NULL	NULL
HOLD	7000	2026-07-29 13:15:37.000000	2026-07-29 13:15:37.000000
