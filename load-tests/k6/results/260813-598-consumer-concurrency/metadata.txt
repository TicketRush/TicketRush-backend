# ── 오버셀 검증 SQL — 상태 무관 검사. 원문을 증적에 남긴다 ─────────────────
# ⚠ 런북 §13.4-(7)(a) 를 그대로 쓰면 검출력이 0 이다. 그 쿼리는
#   booking_status IN ('PENDING','CONFIRMED') 로 거르는데 예매는 5-6분이면 만료되므로
#   검증 시점에 활성 예매가 0건이고, 쿼리가 아무 행도 보지 않은 채 "0행 = 정상" 을 낸다.
#   #554 가 이 문제를 발견해 상태 무관 검사로 바꿔 돌렸다(§9.3). 다만 그 회차는 SQL 원문을
#   남기지 않아 재현이 불가능했다 — 이번에는 원문과 출력을 함께 oversell-<arm>.txt 에 남긴다.
# 각 arm 의 S2 종료 직후, cleanup 전에 실행한다. arm 사이 리셋이 증거를 지운다.
OVERSELL_SQL_F=SELECT b.seat_id, COUNT(*) c FROM booking b WHERE b.performance_id=<PERF_ID> GROUP BY b.seat_id HAVING c > 1;
OVERSELL_SQL_G=SELECT COUNT(*) held_now FROM seat WHERE seat_id=<TARGET_SEAT_ID> AND seat_status='HOLD';
OVERSELL_SQL_H=SELECT COUNT(*) bookings, COUNT(DISTINCT b.seat_id) distinct_seats, COUNT(DISTINCT b.user_id) distinct_users FROM booking b WHERE b.performance_id=<PERF_ID>;
OVERSELL_SQL_I=SELECT b.booking_status, COUNT(*) FROM booking b WHERE b.performance_id=<PERF_ID> GROUP BY 1;
OVERSELL_SQL_J=SELECT seat_id, seat_status, booking_number, version FROM seat WHERE seat_id=<TARGET_SEAT_ID>;
# (f) 0행 = 오버셀 없음 (상태 무관 — 이것이 주 증거다)
# (g) <= 1  = 동시 보유 1건 이하
# (h) 이 회차는 좌석 1석에 VU 100 이 몰리므로 distinct_seats = 1 이 정상이고, bookings 는 그보다 크다.
#     #344 의 1:1:1 과 다르다 — 여기서 보는 것은 seat 중복이 아니라 (f) 가 0행인지다.
# (i) 상태 분포 — (f) 가 왜 상태 무관이어야 하는지의 증적
# (j) version — 엔티티 관리 HOLD 쓰기 횟수. 벌크 만료 UPDATE 는 @Version 을 올리지 않는다(#344 VERSION_NOTE)
OVERSELL_CROSSCHECK=max_over_time(ticketrush_seat_held[8m]) = 1 (#344 OVERSELL_EVIDENCE).
