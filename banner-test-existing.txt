            validRequestBuilder().characterMessage("   ").build(), mainImage, null, null);

@WebMvcSliceTest(BannerController.class)

    mockMvc
        .perform(get(BASE_URL))
        .andExpect(status().isOk())
        .andExpect(jsonPath("$.result[0].id").value(1))
